			function atualizarItemMenu(id_clicado){
				// Para ativar o item pai
				var item_pai_antigo = localStorage.getItem('item_pai_ativo');
				$('#'+item_pai_antigo).removeClass('active');

				var item_novo = $('#'+id_clicado).attr('elemento_raiz');
				$('#'+item_novo).addClass('active');

				localStorage.setItem("item_pai_ativo", item_novo);

				// Para ativar o próprio item
				var item_antigo = localStorage.getItem('item_ativo');
				$('#'+item_antigo).removeClass('active');
				$('#'+item_antigo).parent().removeClass('active');

				$('#'+id_clicado).addClass('active');
				$('#'+id_clicado).parent().addClass('active');
				localStorage.setItem("item_ativo", id_clicado);
			}

			function checkSession() {
				fetch('../sessao_monitoramento_blk/sessao_monitoramento_blk.php')
					.then(response => response.json())
					.then(data => {
					if (data.status === "expired") {
						Swal.fire({
							icon: 'error', 
							title: 'Sessão expirada!',
							text: 'Sua sessão expirou. Você será redirecionado para o login.',
							confirmButtonText: 'OK',
							confirmButtonColor: '#d33',
							willClose: () => {
								let empresa = localStorage.getItem('empresa') || ''; // Evita erro se estiver vazio
								window.location.href = "https://e-grafite.com/portal/?" + empresa;
							}
						});
					} else {
						// Se a sessão ainda estiver ativa, chama novamente a função após 1 minuto
						setTimeout(checkSession, 300000); // 5 minutos
					}
				})
					.catch(error => console.error("Erro ao verificar sessão:", error));
			}

			// Inicia a verificação ao carregar a página
			window.onload = checkSession;
